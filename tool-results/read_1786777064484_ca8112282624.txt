    75→/** Wipe all subcontract test data for the two test orgs (robust to partial failures). */
    76→async function cleanupOrgData() {
    77→  // 1. Fetch all package IDs for the two orgs (and any test-sc prefixed packages,
    78→  //    in case a previous run left data with a different org assignment).
    79→  const packages = await db.subcontractPackage.findMany({
    80→    where: {
    81→      OR: [
    82→        { organizationId: { in: [ORG_A, ORG_B] } },
    83→        { id: { startsWith: 'test-sc-' } },
    84→      ],
    85→    },
    86→    select: { id: true },
    87→  })
    88→  const packageIds = packages.map((p) => p.id)
    89→
    90→  if (packageIds.length > 0) {
    91→    // 2. Delete children by packageId (direct scalar filter — more reliable
    92→    //    than relation filters, which can miss orphaned records).
    93→    await db.quoteScopeCoverage.deleteMany({
    94→      where: { quote: { subcontractPackageId: { in: packageIds } } },
    95→    })
    96→    await db.subcontractQuoteLine.deleteMany({
    97→      where: { quote: { subcontractPackageId: { in: packageIds } } },
    98→    })
    99→    await db.subcontractQuote.deleteMany({
   100→      where: { subcontractPackageId: { in: packageIds } },
   101→    })
   102→    await db.scopeAtom.deleteMany({
   103→      where: { subcontractPackageId: { in: packageIds } },
   104→    })
   105→    await db.subcontractPackageLine.deleteMany({
   106→      where: { subcontractPackageId: { in: packageIds } },
   107→    })
   108→    await db.subcontractPackage.deleteMany({
   109→      where: { id: { in: packageIds } },
   110→    })
   111→  }
   112→
   113→  // 3. Clean up any orphaned subcontract records with test-sc prefix (defensive).
   114→  await db.subcontractQuote.deleteMany({ where: { id: { startsWith: 'test-sc-' } } })
   115→  await db.scopeAtom.deleteMany({ where: { id: { startsWith: 'test-sc-' } } })
   116→
   117→  // 4. Delete the rest by orgId.
   118→  await db.auditLog.deleteMany({ where: { organizationId: { in: [ORG_A, ORG_B] } } })
   119→  await db.commercialException.deleteMany({ where: { organizationId: { in: [ORG_A, ORG_B] } } })
   120→  await db.estimateLine.deleteMany({ where: { id: { in: [LINE_A, LINE_B] } } })
   121→  await db.estimate.deleteMany({ where: { id: { in: [EST_A, EST_B] } } })
   122→  await db.opportunity.deleteMany({ where: { id: { in: [OPP_A, OPP_B] } } })
   123→  await db.client.deleteMany({ where: { id: { in: [CLIENT_A, CLIENT_B] } } })
   124→  await db.user.deleteMany({ where: { id: { in: [USER_A, USER_B] } } })
   125→  await db.organization.deleteMany({ where: { id: { in: [ORG_A, ORG_B] } } })
   126→}

   128→/** Create a package + atoms directly in the DB (for test setup). */
   129→async function createPackageWithAtoms(
   130→  orgId: string,
   131→  packageId: string,
   132→  opportunityId: string,
   133→  atoms: Array<{ id: string; name: string; valueWeight: number }>,
   134→) {
   135→  await db.subcontractPackage.create({
   136→    data: {
   137→      id: packageId,
   138→      organizationId: orgId,
   139→      opportunityId,
   140→      name: `Test Package ${packageId}`,
   141→      executionStrategy: 'subcontract',
   142→    },
   143→  })
   144→  for (const a of atoms) {
   145→    await db.scopeAtom.create({
   146→      data: {
   147→        id: a.id,
   148→        subcontractPackageId: packageId,
   149→        name: a.name,
   150→        valueWeight: a.valueWeight,
   151→      },
   152→    })
   153→  }
   154→}

   156→/** Create a quote + coverages directly in the DB (for test setup). */
   157→async function createQuoteWithCoverages(
   158→  orgId: string,
   159→  packageId: string,
   160→  quoteId: string,
   161→  supplierName: string,
   162→  totalAmount: number,
   163→  coverages: Array<{ atomId: string; status: 'covered' | 'excluded' | 'unstated' }>,
   164→) {
   165→  await db.subcontractQuote.create({
   166→    data: {
   167→      id: quoteId,
   168→      subcontractPackageId: packageId,
   169→      supplierName,
   170→      totalAmount,
   171→      currency: 'GHS',
   172→    },
   173→  })
   174→  for (const c of coverages) {
   175→    await db.quoteScopeCoverage.create({
   176→      data: {
   177→        quoteId,
   178→        scopeAtomId: c.atomId,
   179→        status: c.status,
   180→      },
   181→    })
   182→  }
   183→}
