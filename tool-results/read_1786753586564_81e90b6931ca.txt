   875→   875→/** Build the recipe JSON (CostRecipeLine[]) from a WdSeed, embedding priceObservations. */
   876→   876→function buildRecipeJson(wd: WdSeed): string {
   877→   877→  const recipe: CostRecipeLine[] = wd.recipe.map((r) => ({
   878→   878→    resourceKind: r.kind,
   879→   879→    resourceCode: r.resourceCode,
   880→   880→    resourceName: r.resourceName,
   881→   881→    unit: r.unit,
   882→   882→    quantityPerUnit: r.quantityPerUnit,
   883→   883→    priceObservation: r.unsourced
   884→   884→      ? null
   885→   885→      : {
   886→   886→          price: r.price,
   887→   887→          provenance: r.provenance,
   888→   888→          sourceReference: r.sourceReference,
   889→   889→          observedAt: iso(daysAgo(r.observedAtDaysAgo)),
   890→   890→        },
   891→   891→  }));
   892→   892→  return JSON.stringify(recipe);
   893→   893→}
