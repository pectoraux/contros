     1→     1→/**
     2→     2→ * Tenant-aware repository abstractions.
     3→     3→ *
     4→     4→ * These repositories make unscoped retrieval difficult to express — every
     5→     5→ * method that reads or mutates organization-owned data requires the
     6→     6→ * authenticated organization context.
     7→     7→ *
     8→     8→ * INVARIANT 12: Every organization is isolated from every other organization.
     9→     9→ * A repository must never return an org-owned entity solely from an
    10→    10→ * attacker-supplied ID.
    11→    11→ *
    12→    12→ * P0-3: WorkDefinition / WorkDefinitionVersion / Resource ownership is
    13→    13→ * verified explicitly — not trusted via the EstimateLine relation.
    14→    14→ */

    22→    22→  /** Get an estimate scoped to the authenticated organization. */
    23→    23→  async getForOrganization(orgId: string, estimateId: string) {
    24→    24→    return db.estimate.findFirst({
    25→    25→      where: { id: estimateId, organizationId: orgId },
    26→    26→    })
    27→    27→  },
    28→    28→
    29→    29→  /**
    30→    30→   * Get the full revision context for an estimate: the estimate, its lines,
    31→    31→   * work definitions, versions, execution segments, and existing revisions.
    32→    32→   *
    33→    33→   * P0-3: WorkDefinition/WDV/Resource ownership is enforced AT THE REPOSITORY
    34→    34→   * LEVEL — the query only loads WDs/WDVs/priceObservations that belong to the
    35→    35→   * requesting organization. If a line references a cross-tenant WD, the WD/WDV
    36→    36→   * will be null in the result. The service checks for this and rejects.
    37→    37→   */
    38→    38→  async getRevisionContext(orgId: string, estimateId: string) {
    39→    39→    return db.estimate.findFirst({
    40→    40→      where: { id: estimateId, organizationId: orgId },
    41→    41→      include: {
    42→    42→        lines: {
    43→    43→          include: {
    44→    44→            // Only load WD if it belongs to the same org (1:1 relation with where filter)
    45→    45→            workDefinition: {
    46→    46→              where: { organizationId: orgId },
    47→    47→            },
    48→    48→            // Only load WDV if its parent WD belongs to the same org
    49→    49→            workDefinitionVersion: {
    50→    50→              where: {
    51→    51→                workDefinition: { organizationId: orgId },
    52→    52→              },
    53→    53→              include: {
    54→    54→                // Only load price observations for resources in the same org
    55→    55→                priceObservations: {
    56→    56→                  where: {
    57→    57→                    resource: { organizationId: orgId },
    58→    58→                  },
    59→    59→                },
    60→    60→              },
    61→    61→            },
    62→    62→            executionSegments: true,
    63→    63→          },
    64→    64→        },
    65→    65→        revisions: { orderBy: { revisionNo: 'desc' }, take: 1 },
    66→    66→      },
    67→    67→    })
    68→    68→  },
    69→    69→
    70→    70→  /**
    71→    71→   * Get an estimate line with full pricing graph, tenant-scoped.
    72→    72→   * P0-3: WD/WDV/Resource ownership enforced at the repository level.
    73→    73→   */
    74→    74→  async getLineForOrganization(
    75→    75→    orgId: string,
    76→    76→    estimateId: string,
    77→    77→    lineId: string,
    78→    78→  ) {
    79→    79→    return db.estimateLine.findFirst({
    80→    80→      where: {
    81→    81→        id: lineId,
    82→    82→        estimateId,
    83→    83→        estimate: { organizationId: orgId },
    84→    84→      },
    85→    85→      include: {
    86→    86→        workDefinition: {
    87→    87→          where: { organizationId: orgId },
    88→    88→        },
    89→    89→        workDefinitionVersion: {
    90→    90→          where: {
    91→    91→            workDefinition: { organizationId: orgId },
    92→    92→          },
    93→    93→          include: {
    94→    94→            priceObservations: {
    95→    95→              where: {
    96→    96→                resource: { organizationId: orgId },
    97→    97→              },
    98→    98→            },
    99→    99→          },
   100→   100→        },
   101→   101→        estimate: true,
   102→   102→        scopeItem: true,
   103→   103→        executionSegments: true,
   104→   104→      },
   105→   105→    })
   106→   106→  },
