   860→/** Build the recipe JSON (CostRecipeLine[]) from a WdSeed, embedding priceObservations. */
   861→function buildRecipeJson(wd: WdSeed): string {
   862→  const recipe: CostRecipeLine[] = wd.recipe.map((r) => ({
   863→    resourceKind: r.kind,
   864→    resourceCode: r.resourceCode,
   865→    resourceName: r.resourceName,
   866→    unit: r.unit,
   867→    quantityPerUnit: r.quantityPerUnit,
   868→    priceObservation: r.unsourced
   869→      ? null
   870→      : {
   871→          price: r.price,
   872→          provenance: r.provenance,
   873→          sourceReference: r.sourceReference,
   874→          observedAt: iso(daysAgo(r.observedAtDaysAgo)),
   875→        },
   876→  }));
   877→  return JSON.stringify(recipe);
   878→}

   880→async function seedWorkDefinitions(ctx: SeedCtx): Promise<void> {
   881→  console.log('• Seeding Work Definitions + Resources + Price Observations...');
   882→
   883→  // Track resources so we don't create duplicates (cement appears in multiple WDs).
   884→  const createdResources = new Set<string>();
   885→
   886→  for (const wd of WORK_DEFINITIONS) {
   887→    // 1. Create the WorkDefinition row (without currentVersionId yet).
   888→    await prisma.workDefinition.create({
   889→      data: {
   890→        id: wd.wdId,
   891→        organizationId: ctx.orgId,
   892→        code: wd.code,

   897→        approvalState: 'approved',
   898→      },
   899→    });
   900→    ctx.wdIds.set(wd.wdId, wd.wdId);
   901→
   902→    // 2. Build the recipe JSON.
   903→    const costRecipeJson = buildRecipeJson(wd);
   904→
   905→    // 3. Create the WorkDefinitionVersion (approved).
   906→    await prisma.workDefinitionVersion.create({
   907→      data: {
   908→        id: wd.versionId,
   909→        workDefinitionId: wd.wdId,
   910→        version: 1,
   911→        measurementRule: `per ${wd.unit}`,
   912→        costRecipeJson,
   913→        productivityRule: wd.productivity,
   914→        crewComposition: 'Standard crew (see method statement)',
   915→        equipment: wd.recipe.find((r) => r.kind === 'plant')?.resourceName ?? 'Hand tools',
   916→        wastage: wd.wastage,
   917→        sequencing: 'Per method statement; sequence to programme.',
   918→        methodStatementFragment: wd.methodStatementFragment,
   919→        hazardsJson: JSON.stringify(wd.hazards),
   920→        controlsJson: JSON.stringify(wd.controls),
   921→        qualityChecklistJson: JSON.stringify(wd.qualityChecklist),
   922→        requiredPPE: wd.ppe,
   923→        requiredPermits: wd.permits,
   924→        subcontractability: wd.subcontractability,
   925→        commonAssumptions: wd.commonAssumptions,
   926→        commonExclusions: wd.commonExclusions,
   927→        approvalState: 'approved',
   928→        approvedAt: daysAgo(45),
   929→        approvedById: ctx.userIds.kwesi,
   930→      },
   931→    });
   932→    ctx.wdVersionIds.set(wd.wdId, wd.versionId);
   933→
   934→    // 4. Link WD.currentVersionId.
   935→    await prisma.workDefinition.update({
   936→      where: { id: wd.wdId },
   937→      data: { currentVersionId: wd.versionId },
   938→    });
   939→
   940→    // 5. Create Resource + ResourcePriceObservation for each recipe line.
   941→    for (const r of wd.recipe) {
   942→      if (!createdResources.has(r.resourceCode)) {
   943→        const resourceId = `res-${r.resourceCode.toLowerCase().replace(/_/g, '-')}`;
   944→        await prisma.resource.create({
   945→          data: {
   946→            id: resourceId,
   947→            organizationId: ctx.orgId,
   948→            code: r.resourceCode,
   949→            name: r.resourceName,
   950→            unit: r.unit,
   951→            kind: r.kind,
   952→            currency: 'GHS',
   953→            region: 'Greater Accra',
   954→          },
   955→        });
   956→        ctx.resourceIds.set(r.resourceCode, resourceId);
   957→        createdResources.add(r.resourceCode);
   958→      }
   959→
   960→      // Create the price observation (skip if recipe line is unsourced).
   961→      if (!r.unsourced) {
   962→        const resourceId = ctx.resourceIds.get(r.resourceCode)!;
   963→        await prisma.resourcePriceObservation.create({
   964→          data: {
   965→            id: `rpo-${wd.versionId}-${r.resourceCode.toLowerCase().replace(/_/g, '-')}`,
   966→            resourceId,
   967→            workDefinitionVersionId: wd.versionId,
   968→            price: r.price,
   969→            currency: 'GHS',
   970→            provenance: r.provenance,
   971→            sourceReference: r.sourceReference ?? null,
   972→            observedAt: daysAgo(r.observedAtDaysAgo),
   973→            recordedById: ctx.userIds.abena,
   974→          },
   975→        });
   976→      }
   977→    }
   978→  }
   979→
   980→  console.log(
   981→    `  ✓ ${WORK_DEFINITIONS.length} work definitions, ${ctx.resourceIds.size} resources, ${await prisma.resourcePriceObservation.count()} price observations.`,
   982→  );
   983→}

  1407→/** Build a human-readable provenance summary from a PricingBreakdown. */
  1408→function buildProvenanceSummary(b: PricingBreakdown): string {
  1409→  if (b.provenance.length === 0) {
  1410→    if (b.unsourcedResources.length > 0) {
  1411→      return `UNSOURCED: ${b.unsourcedResources.join(', ')}`;
  1412→    }
  1413→    return 'No price provenance recorded.';
  1414→  }
  1415→  return b.provenance
  1416→    .map((p) => {
  1417→      const refPart = p.sourceReference ? ` #${p.sourceReference}` : '';
  1418→      return `${p.resourceName}: ${p.provenance}${refPart} @ ${formatGHS(p.price)}`;
  1419→    })
  1420→    .join('; ');
  1421→}

  1423→/** Compute pricing + confidence for a line and persist it. Returns the line id + breakdown. */
  1424→async function createEstimateLine(
  1425→  ctx: SeedCtx,
  1426→  estimateId: string,
  1427→  lineSeed: EstimateLineSeed,
  1428→  scopeCompleteness: number,
  1429→  lineIndex: number,
  1430→  estimateCreatedAt: Date,
  1431→  lineIdPrefix: string,
  1432→): Promise<LinePricingResult> {
  1433→  const wdId = ctx.wdIds.get(lineSeed.wdId)!;
  1434→  const versionId = ctx.wdVersionIds.get(lineSeed.wdId)!;
  1435→
  1436→  // Look up the WD version (we need name, unit, wastage, productivity, costRecipeJson).
  1437→  const wdv = await prisma.workDefinitionVersion.findUnique({
  1438→    where: { id: versionId },
  1439→    select: {
  1440→      version: true,
  1441→      wastage: true,
  1442→      productivityRule: true,
  1443→      costRecipeJson: true,
  1444→      approvalState: true,
  1445→      workDefinition: { select: { name: true, unit: true } },
  1446→    },
  1447→  });
  1448→  if (!wdv) throw new Error(`WDV ${versionId} not found`);
  1449→
  1450→  // Build the PricingInput.
  1451→  const breakdown = priceLine({
  1452→    workDefinitionVersion: {
  1453→      id: versionId,
  1454→      name: wdv.workDefinition.name,
  1455→      version: wdv.version,
  1456→      unit: wdv.workDefinition.unit,
  1457→      wastage: wdv.wastage,
  1458→      productivityRule: wdv.productivityRule ?? undefined,
  1459→      costRecipeJson: wdv.costRecipeJson,
  1460→    },
  1461→    quantity: lineSeed.quantity,
  1462→    executionStrategy: lineSeed.executionStrategy,
  1463→    overheadPct: 0.1,
  1464→    profitPct: 0.12,
  1465→    contingencyPct: 0.05,
  1466→    subcontractQuote: lineSeed.subcontractQuote ?? null,
  1467→  });
  1468→
  1469→  // Compute confidence. Use the oldest observedAt in the provenance (or the recipe)
  1470→  // for the freshness factor.
  1471→  let observedAtForConfidence: string | null = null;
  1472→  let provenanceForConfidence: string | null = null;
  1473→  if (breakdown.provenance.length > 0) {
  1474→    // Use the oldest observation (most conservative freshness).
  1475→    const sorted = [...breakdown.provenance].sort(
  1476→      (a, b) => Date.parse(a.observedAt) - Date.parse(b.observedAt),
  1477→    );
  1478→    observedAtForConfidence = sorted[0].observedAt;
  1479→    provenanceForConfidence = sorted[0].provenance;
  1480→  }
  1481→
  1482→  const confidenceResult = computeConfidence({
  1483→    observedAt: observedAtForConfidence,
  1484→    referenceDate: iso(estimateCreatedAt),
  1485→    priceProvenance: provenanceForConfidence,
  1486→    workDefinitionApprovalState: wdv.approvalState,
  1487→    scopeCompleteness,
  1488→    executionStrategy: lineSeed.executionStrategy,
  1489→    subcontractQuotePresent:
  1490→      lineSeed.executionStrategy === 'subcontract'
  1491→        ? Boolean(lineSeed.subcontractQuote)
  1492→        : null,
  1493→    productivityRule: wdv.productivityRule,
  1494→    quantity: lineSeed.quantity,
  1495→    unit: wdv.workDefinition.unit,
  1496→  });
  1497→
  1498→  const lineId = `${lineIdPrefix}-${lineIndex + 1}`;
  1499→  await prisma.estimateLine.create({
  1500→    data: {
  1501→      id: lineId,
  1502→      estimateId,
  1503→      scopeItemId: lineSeed.scopeItemId,
  1504→      workDefinitionId: wdId,
  1505→      workDefinitionVersionId: versionId,
  1506→      description: lineSeed.description,
  1507→      quantity: lineSeed.quantity,
  1508→      unit: wdv.workDefinition.unit,
  1509→      executionStrategy: lineSeed.executionStrategy,
  1510→      materialCost: breakdown.material,
  1511→      labourCost: breakdown.labour,
  1512→      plantCost: breakdown.plant,
  1513→      subcontractCost: breakdown.subcontract,
  1514→      directCost: breakdown.directCost,
  1515→      projectCost: breakdown.projectCost,
  1516→      riskCost: breakdown.riskCost,
  1517→      overheadCost: breakdown.overhead,
  1518→      profitCost: breakdown.profit,
  1519→      sellPrice: breakdown.sellPrice,
  1520→      unitRate: breakdown.unitRate,
  1521→      marginPct: breakdown.marginPct,
  1522→      confidence: confidenceResult.score,
  1523→      provenanceSummary: buildProvenanceSummary(breakdown),
  1524→      isUnsourced: breakdown.unsourced,
  1525→      unsourcedRationale: breakdown.unsourced
  1526→        ? (lineSeed.unsourcedRationale ?? `Unsourced resources: ${breakdown.unsourcedResources.join(', ')}`)
  1527→        : null,
  1528→      acknowledged: breakdown.unsourced
  1529→        ? (lineSeed.acknowledged ?? false)
  1530→        : true,
  1531→    },
  1532→  });
  1533→
  1534→  return {
  1535→    lineId,
  1536→    breakdown,
  1537→    confidence: confidenceResult.score,
  1538→    sellPrice: breakdown.sellPrice,
  1539→  };
  1540→}
