     1→/**
     2→ * Tenant-aware repository abstractions.
     3→ *
     4→ * These repositories make unscoped retrieval difficult to express — every
     5→ * method that reads or mutates organization-owned data requires the
     6→ * authenticated organization context.
     7→ *
     8→ * INVARIANT 12: Every organization is isolated from every other organization.
     9→ * A repository must never return an org-owned entity solely from an
    10→ * attacker-supplied ID.
    11→ *
    12→ * P0-3: WorkDefinition / WorkDefinitionVersion / Resource ownership is
    13→ * verified explicitly — not trusted via the EstimateLine relation.
    14→ */
    15→
    16→import { db } from '@/lib/db'
    17→import type { RequestContext } from '@/lib/context'

    19→// ─── Estimate Repository ────────────────────────────────────────────────────
    20→
    21→export const estimateRepository = {
    22→  /** Get an estimate scoped to the authenticated organization. */
    23→  async getForOrganization(orgId: string, estimateId: string) {
    24→    return db.estimate.findFirst({
    25→      where: { id: estimateId, organizationId: orgId },
    26→    })
    27→  },
    28→
    29→  /**
    30→   * Get the full revision context for an estimate: the estimate, its lines,
    31→   * work definitions, versions, execution segments, and existing revisions.
    32→   *
    33→   * P0-3: WorkDefinition/WDV/Resource ownership is enforced AT THE REPOSITORY
    34→   * LEVEL — the query only loads WDs/WDVs/priceObservations that belong to the
    35→   * requesting organization. If a line references a cross-tenant WD, the WD/WDV
    36→   * will be null in the result. The service checks for this and rejects.
    37→   */
    38→  async getRevisionContext(orgId: string, estimateId: string) {
    39→    return db.estimate.findFirst({
    40→      where: { id: estimateId, organizationId: orgId },
    41→      include: {
    42→        lines: {
    43→          include: {
    44→            // Only load WD if it belongs to the same org (1:1 relation with where filter)
    45→            workDefinition: {
    46→              where: { organizationId: orgId },
    47→            },
    48→            // Only load WDV if its parent WD belongs to the same org
    49→            workDefinitionVersion: {
    50→              where: {
    51→                workDefinition: { organizationId: orgId },
    52→              },
    53→              include: {
    54→                // Only load price observations for resources in the same org
    55→                priceObservations: {
    56→                  where: {
    57→                    resource: { organizationId: orgId },
    58→                  },
    59→                },
    60→              },
    61→            },
    62→            executionSegments: true,
    63→          },
    64→        },
    65→        revisions: { orderBy: { revisionNo: 'desc' }, take: 1 },
    66→      },
    67→    })
    68→  },
    69→
    70→  /**
    71→   * Get an estimate line with full pricing graph, tenant-scoped.
    72→   * P0-3: WD/WDV/Resource ownership enforced at the repository level.
    73→   */
    74→  async getLineForOrganization(
    75→    orgId: string,
    76→    estimateId: string,
    77→    lineId: string,
    78→  ) {
    79→    return db.estimateLine.findFirst({
    80→      where: {
    81→        id: lineId,
    82→        estimateId,
    83→        estimate: { organizationId: orgId },
    84→      },
    85→      include: {
    86→        workDefinition: {
    87→          where: { organizationId: orgId },
    88→        },
    89→        workDefinitionVersion: {
    90→          where: {
    91→            workDefinition: { organizationId: orgId },
    92→          },
    93→          include: {
    94→            priceObservations: {
    95→              where: {
    96→                resource: { organizationId: orgId },
    97→              },
    98→            },
    99→          },
   100→        },
   101→        estimate: true,
   102→        scopeItem: true,
   103→        executionSegments: true,
   104→      },
   105→    })
   106→  },
   107→
   108→  /** Update an estimate line — tenant-scoped via the estimate relation. */
   109→  async updateLine(
   110→    orgId: string,
   111→    lineId: string,
   112→    data: Record<string, unknown>,
   113→  ) {
   114→    const line = await db.estimateLine.findFirst({
   115→      where: { id: lineId, estimate: { organizationId: orgId } },
   116→      select: { id: true },
   117→    })
   118→    if (!line) return null
   119→    return db.estimateLine.update({ where: { id: lineId }, data })
   120→  },
   121→}

   297→  /**
   298→   * Create a quote, verifying package ownership.
   299→   * Verifies: package → opportunity → org.
   300→   * Returns null if the package doesn't exist OR belongs to another org.
   301→   */
   302→  async createForPackage(
   303→    orgId: string,
   304→    packageId: string,
   305→    data: {
   306→      supplierName: string
   307→      totalAmount: number
   308→      currency: string

   320→        opportunity: { organizationId: orgId },

   325→    return db.subcontractQuote.create({

   327→        id: data.quoteId,
   328→        subcontractPackageId: packageId,
   329→        supplierName: data.supplierName,
   330→        totalAmount: data.totalAmount,

   335→      },
   336→    })
   337→  },
   338→
   339→  /**
   340→   * Create a quote within a transaction, verifying package ownership.
   341→   *
   342→   * Used by SubcontractService.recordQuote() so that the quote creation and

   346→   * Returns null if the package doesn't exist OR belongs to another org.

   354→      totalAmount: number

   359→      quoteId?: string

   365→        id: packageId,
   366→        organizationId: orgId,
   367→        opportunity: { organizationId: orgId },
   368→      },
   369→      select: { id: true },
   370→    })

   374→        id: data.quoteId,
   375→        subcontractPackageId: packageId,
   376→        supplierName: data.supplierName,

   379→        exclusionsJson: data.exclusionsJson,

   383→    })
   384→  },

   386→  /**
   387→   * Update quote status (selected/rejected), tenant-scoped.
   388→   * Verifies: quote → package → opportunity → org.
   389→   * Returns null if the quote doesn't exist OR belongs to another org.
   390→   */
   391→  async updateStatus(orgId: string, quoteId: string, status: string) {

   394→        id: quoteId,
   395→        subcontractPackage: {
   396→          opportunity: { organizationId: orgId },
   397→        },
   398→      },
   399→      select: { id: true },
   400→    })
   401→    if (!quote) return null
   402→    return db.subcontractQuote.update({
   403→      where: { id: quoteId },
   404→      data: { status },
   405→    })
   406→  },
   407→
   408→  /**
   409→   * Update quote status within a transaction (used by selectQuote).

   412→  async updateStatusInTransaction(
