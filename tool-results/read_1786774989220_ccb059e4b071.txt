   455→// P0-7 fix: scope reconciliation based on structured scope atoms, not substring.
   456→// A SubcontractPackage defines required ScopeAtoms (e.g. manufacture, delivery,
   457→// installation). Each quote explicitly covers/excludes each atom.
   458→model ScopeAtom {
   459→  id                   String   @id @default(cuid())
   460→  subcontractPackageId String
   461→  name                 String   // e.g. "installation"
   462→  description          String?
   463→  // P0-1 fix: economic weight for this atom (0..1). Enables economic coverage
   464→  // = sum(covered atom weights) / sum(all atom weights), distinct from semantic
   465→  // coverage = covered atoms / total atoms. If all weights are equal (or 0),
   466→  // economic coverage falls back to semantic coverage.
   467→  valueWeight          Float    @default(0)
   468→  createdAt            DateTime @default(now())
   469→
   470→  subcontractPackage   SubcontractPackage @relation(fields: [subcontractPackageId], references: [id])
